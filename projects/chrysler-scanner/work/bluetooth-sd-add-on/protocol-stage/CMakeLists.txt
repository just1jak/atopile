cmake_minimum_required(VERSION 3.16)
project(chrysler_logger_protocol LANGUAGES CXX)
add_library(chrysler_protocol src/protocol.cpp)
target_include_directories(chrysler_protocol PUBLIC include)
target_compile_features(chrysler_protocol PUBLIC cxx_std_17)
target_compile_options(chrysler_protocol PRIVATE -Wall -Wextra -Wpedantic)
add_executable(protocol_test tests/protocol_test.cpp)
target_link_libraries(protocol_test PRIVATE chrysler_protocol)
enable_testing()
add_test(NAME protocol_test COMMAND protocol_test)
